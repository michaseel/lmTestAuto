        const app = {
            data: {
                columns: ['backlog', 'inProgress', 'review', 'done'],
                cards: [],
                pendingDeleteId: null
            },

            // Initialize the app
            init() {
                this.loadData();
                this.render();
            },

            // --- Data Management ---

            loadData() {
                const stored = localStorage.getItem('kanban-board-v1');
                if (stored) {
                    this.data.cards = JSON.parse(stored);
                } else {
                    // Default dummy data if nothing exists
                    this.data.cards = [
                        { id: 'c1', title: 'Design System Setup', desc: 'Define colors and typography.', status: 'backlog' },
                        { id: 'c2', title: 'Implement Login', desc: 'Use OAuth2.', status: 'inProgress' },
                        { id: 'c3', title: 'Review PR #42', desc: 'Check API integration.', status: 'review' },
                        { id: 'c4', title: 'Deploy to Staging', desc: 'Verify all features.', status: 'done' }
                    ];
                }
            },

            saveData() {
                localStorage.setItem('kanban-board-v1', JSON.stringify(this.data.cards));
                this.renderCounts();
            },

            resetBoard() {
                if(confirm("Are you sure you want to reset the board to default state?")) {
                    localStorage.removeItem('kanban-board-v1');
                    this.loadData();
                    this.render();
                }
            },

            // --- Rendering ---

            render() {
                // Clear all columns
                this.data.columns.forEach(col => {
                    document.getElementById(`col-${col}`).innerHTML = '';
                });

                // Populate cards
                this.data.cards.forEach(card => {
                    const cardEl = this.createCardElement(card);
                    document.getElementById(`col-${card.status}`).appendChild(cardEl);
                });

                this.renderCounts();
            },

            createCardElement(card) {
                const div = document.createElement('div');
                div.className = 'card bg-white p-4 rounded-xl shadow-sm border border-gray-100 cursor-grab active:cursor-grabbing relative group';
                div.draggable = true;
                div.id = card.id;

                // Drag Events
                div.addEventListener('dragstart', (e) => this.handleDragStart(e, card.id));
                div.addEventListener('dragend', (e) => this.handleDragEnd(e));

                // Edit on double click (title only for simplicity)
                const titleSpan = document.createElement('div');
                titleSpan.contentEditable = true;
                titleSpan.className = 'font-semibold text-gray-800 mb-1 outline-none focus:bg-gray-50 rounded px-1';
                titleSpan.innerText = card.title;
                titleSpan.addEventListener('blur', (e) => this.updateCardTitle(card.id, e.target.innerText));
                titleSpan.addEventListener('keydown', (e) => {
                    if (e.key === 'Enter') {
                        e.preventDefault();
                        e.target.blur();
                    }
                });

                const descEl = document.createElement('p');
                descEl.className = 'text-sm text-gray-500 line-clamp-2';
                descEl.innerText = card.desc || '';

                // Delete Button (appears on hover)
                const deleteBtn = document.createElement('button');
                deleteBtn.className = 'absolute top-2 right-2 text-gray-300 hover:text-red-500 opacity-0 group-hover:opacity-100 transition-opacity p-1';
                deleteBtn.innerHTML = '<svg class="w-4 h-4" fill="none" stroke="currentColor" viewBox="0 0 24 24"><path stroke-linecap="round" stroke-linejoin="round" stroke-width="2" d="M6 18L18 6M6 6l12 12"></path></svg>';
                deleteBtn.onclick = (e) => {
                    e.stopPropagation(); // prevent drag start
                    this.openDeleteModal(card.id);
                };

                div.appendChild(titleSpan);
                div.appendChild(descEl);
                div.appendChild(deleteBtn);

                return div;
            },

            renderCounts() {
                this.data.columns.forEach(col => {
                    const count = this.data.cards.filter(c => c.status === col).length;
                    document.getElementById(`count-${col}`).innerText = count;
                });
            },

            // --- Actions ---

            openModal(status, editId = null) {
                const modal = document.getElementById('cardModal');
                const titleInput = document.getElementById('inputTitle');
                const descInput = document.getElementById('inputDesc');
                const editIdInput = document.getElementById('editCardId');
                const statusInput = document.getElementById('editCardStatus');
                const modalTitle = document.getElementById('modalTitle');

                modal.classList.remove('hidden');
                modal.classList.add('flex');
                titleInput.value = '';
                descInput.value = '';

                if (editId) {
                    const card = this.data.cards.find(c => c.id === editId);
                    modalTitle.innerText = 'Edit Card';
                    titleInput.value = card.title;
                    descInput.value = card.desc;
                    editIdInput.value = card.id;
                    statusInput.value = card.status;
                } else {
                    modalTitle.innerText = 'Add New Card';
                    editIdInput.value = '';
                    statusInput.value = status;
                }
                titleInput.focus();
            },

            closeModal() {
                const modal = document.getElementById('cardModal');
                modal.classList.add('hidden');
                modal.classList.remove('flex');
            },

            saveCard() {
                const title = document.getElementById('inputTitle').value.trim();
                const desc = document.getElementById('inputDesc').value.trim();
                const editId = document.getElementById('editCardId').value;
                const status = document.getElementById('editCardStatus').value;

                if (!title) {
                    alert('Please enter a title');
                    return;
                }

                if (editId) {
                    // Update existing
                    const cardIndex = this.data.cards.findIndex(c => c.id === editId);
                    if (cardIndex > -1) {
                        this.data.cards[cardIndex].title = title;
                        this.data.cards[cardIndex].desc = desc;
                    }
                } else {
                    // Create new
                    const newCard = {
                        id: 'c' + Date.now(),
                        title: title,
                        desc: desc,
                        status: status
                    };
                    this.data.cards.push(newCard);
                }

                this.saveData();
                this.render();
                this.closeModal();
            },

            updateCardTitle(id, newTitle) {
                const card = this.data.cards.find(c => c.id === id);
                if (card && newTitle.trim() !== '') {
                    card.title = newTitle.trim();
                    this.saveData();
                } else {
                    this.render(); // Revert if empty
                }
            },

            // --- Delete Logic ---

            openDeleteModal(id) {
                this.pendingDeleteId = id;
                const modal = document.getElementById('deleteModal');
                modal.classList.remove('hidden');
                modal.classList.add('flex');
            },

            closeDeleteModal() {
                const modal = document.getElementById('deleteModal');
                modal.classList.add('hidden');
                modal.classList.remove('flex');
                this.pendingDeleteId = null;
            },

            confirmDelete() {
                if (this.pendingDeleteId) {
                    this.data.cards = this.data.cards.filter(c => c.id !== this.pendingDeleteId);
                    this.saveData();
                    this.render();
                    this.closeDeleteModal();
                }
            },

            // --- Drag and Drop ---

            handleDragStart(e, id) {
                e.dataTransfer.setData("text/plain", id);
                e.dataTransfer.effectAllowed = "move";
                e.target.classList.add('opacity-50', 'scale-95');
                setTimeout(() => e.target.classList.add('hidden'), 0); // Hide original momentarily
            },

            handleDragEnd(e) {
                e.target.classList.remove('opacity-50', 'scale-95', 'hidden');
                // Clean up any drag enter styles
                document.querySelectorAll('.kanban-column').forEach(col => {
                    col.classList.remove('bg-gray-100', 'border-dashed');
                });
            },

            allowDrop(e) {
                e.preventDefault();
            },

            handleDragEnter(e) {
                e.preventDefault();
                const col = e.currentTarget;
                col.classList.add('bg-gray-100', 'border-dashed');
            },

            handleDragLeave(e) {
                const col = e.currentTarget;
                // Only remove if we are actually leaving the column (not just entering a child)
                if (!col.contains(e.relatedTarget)) {
                    col.classList.remove('bg-gray-100', 'border-dashed');
                }
            },

            handleDrop(e, targetStatus) {
                e.preventDefault();
                const col = e.currentTarget;
                col.classList.remove('bg-gray-100', 'border-dashed');

                const id = e.dataTransfer.getData("text/plain");
                const cardIndex = this.data.cards.findIndex(c => c.id === id);
                
                if (cardIndex > -1 && this.data.cards[cardIndex].status !== targetStatus) {
                    // Update status
                    this.data.cards[cardIndex].status = targetStatus;
                    this.saveData();
                    this.render();

                    // Check if moved to Done
                    if (targetStatus === 'done') {
                        this.triggerConfetti();
                    }
                }
            },

            // Visual flair for completing a task
            triggerConfetti() {
                // Simple CSS animation on the Done column
                const col = document.getElementById('col-done');
                col.animate([
                    { transform: 'scale(1)' },
                    { transform: 'scale(1.02)' },
                    { transform: 'scale(1)' }
                ], {
                    duration: 300,
                    easing: 'ease-out'
                });
            }
        };

        // Start the app
        document.addEventListener('DOMContentLoaded', () => {
            app.init();
        });
    </script>
</body>
</html>
