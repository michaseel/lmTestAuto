```html
<!DOCTYPE html>
<html lang="en">
<head>
    <meta charset="UTF-8">
    <meta name="viewport" content="width=device-width, initial-scale=1.0">
    <title>Vanilla Kanban Board</title>
    <script src="https://cdn.tailwindcss.com"></script>
    <script src="https://cdn.jsdelivr.net/npm/canvas-confetti@1.9.2/dist/confetti.browser.min.js"></script>
    <script>
        // Configure Tailwind for dark mode and custom colors
        tailwind.config = {
            darkMode: 'class',
            theme: {
                extend: {
                    colors: {
                        primary: '#6366f1', // Indigo
                        secondary: '#8b5cf6', // Violet
                        success: '#10b981', // Emerald
                        warning: '#f59e0b', // Amber
                        danger: '#ef4444', // Red
                    }
                }
            }
        }
    </script>
    <style type="text/tailwindcss">
        @layer components {
            .kanban-card {
                @apply p-4 rounded-lg shadow-md transition-all duration-200 hover:shadow-lg border border-gray-200 dark:border-gray-700;
            }
            .kanban-column {
                @apply flex flex-col h-full w-72 rounded-xl p-4 bg-gray-50 dark:bg-gray-800 shadow;
            }
            .kanban-header {
                @apply flex justify-between items-center mb-4;
            }
            .kanban-title {
                @apply font-bold text-lg text-gray-800 dark:text-white;
            }
            .kanban-count {
                @apply bg-gray-200 dark:bg-gray-700 text-gray-800 dark:text-white rounded-full px-2 py-1 text-xs font-semibold;
            }
            .kanban-add-btn {
                @apply bg-indigo-500 hover:bg-indigo-600 text-white px-3 py-1 rounded-md text-sm font-medium transition-colors duration-200;
            }
            .kanban-card-content {
                @apply text-gray-700 dark:text-gray-300 mb-3;
            }
            .kanban-card-actions {
                @apply flex justify-end space-x-2;
            }
            .kanban-delete-btn {
                @apply text-red-500 hover:text-red-700 transition-colors duration-200;
            }
            .kanban-edit-btn {
                @apply text-indigo-500 hover:text-indigo-700 transition-colors duration-200;
            }
        }
    </style>
</head>
<body class="bg-gray-100 dark:bg-gray-900 min-h-screen p-4 md:p-8">
    <div class="max-w-7xl mx-auto">
        <h1 class="text-3xl font-bold text-center mb-8 text-gray-800 dark:text-white">Vanilla Kanban Board</h1>
        
        <!-- Kanban Board Container -->
        <div id="kanban-board" class="flex flex-wrap justify-center gap-6">
            <!-- Columns will be dynamically added here -->
        </div>
    </div>

    <script>
        // Define the Kanban board namespace to avoid conflicts
        const KANBAN_NAMESPACE = 'kanban_board_v1';
        
        // Define columns with their titles and IDs
        const COLUMNS = [
            { id: 'backlog', title: 'Backlog' },
            { id: 'in_progress', title: 'In Progress' },
            { id: 'review', title: 'Review' },
            { id: 'done', title: 'Done' }
        ];

        // DOM elements
        const boardContainer = document.getElementById('kanban-board');
        
        // Card editing state
        let currentEditingCard = null;
        let currentEditingInput = null;

        // Initialize the Kanban board
        function initKanbanBoard() {
            // Load saved data from localStorage or create dummy data if none exists
            const savedData = localStorage.getItem(KANBAN_NAMESPACE);
            let cards = [];
            
            if (savedData) {
                try {
                    cards = JSON.parse(savedData);
                } catch (e) {
                    console.error('Failed to parse saved data:', e);
                    cards = [];
                }
            }

            // If no cards exist, create dummy data
            if (cards.length === 0) {
                cards = [
                    { id: '1', content: 'Create project structure', columnId: 'backlog' },
                    { id: '2', content: 'Implement drag and drop', columnId: 'backlog' },
                    { id: '3', content: 'Design UI components', columnId: 'backlog' },
                    { id: '4', content: 'Set up local storage', columnId: 'in_progress' },
                    { id: '5', content: 'Add card deletion confirmation', columnId: 'in_progress' },
                    { id: '6', content: 'Style with Tailwind', columnId: 'review' },
                    { id: '7', content: 'Add confetti animation', columnId: 'done' },
                    { id: '8', content: 'Implement in-place editing', columnId: 'done' }
                ];
                saveToLocalStorage(cards);
            }

            // Render all columns and cards
            renderBoard(cards);
        }

        // Save data to localStorage with namespace
        function saveToLocalStorage(cards) {
            try {
                localStorage.setItem(KANBAN_NAMESPACE, JSON.stringify(cards));
            } catch (e) {
                console.error('Failed to save to localStorage:', e);
            }
        }

        // Render the entire Kanban board
        function renderBoard(cards) {
            // Clear existing content
            boardContainer.innerHTML = '';

            // Create columns and populate them with cards
            COLUMNS.forEach(column => {
                const columnElement = createColumnElement(column);
                
                // Filter cards for this column
                const columnCards = cards.filter(card => card.columnId === column.id);
                
                // Add each card to the column
                columnCards.forEach(card => {
                    const cardElement = createCardElement(card);
                    columnElement.querySelector('.kanban-column-content').appendChild(cardElement);
                });
                
                // Add "Add card" button to the column
                const addCardButton = createAddCardButton(column.id);
                columnElement.querySelector('.kanban-column-content').appendChild(addCardButton);
                
                // Add the column to the board
                boardContainer.appendChild(columnElement);
            });
            
            // Update card counts for each column
            updateCardCounts(cards);
        }

        // Create a column DOM element
        function createColumnElement(column) {
            const columnElement = document.createElement('div');
            columnElement.className = 'kanban-column';
            columnElement.id = `column-${column.id}`;
            
            // Column header with title and count
            const header = document.createElement('div');
            header.className = 'kanban-header';
            
            const title = document.createElement('h2');
            title.className = 'kanban-title';
            title.textContent = column.title;
            
            const count = document.createElement('span');
            count.className = 'kanban-count';
            count.id = `count-${column.id}`;
            
            header.appendChild(title);
            header.appendChild(count);
            
            // Column content container
            const content = document.createElement('div');
            content.className = 'kanban-column-content flex flex-col space-y-3';
            
            columnElement.appendChild(header);
            columnElement.appendChild(content);
            
            return columnElement;
        }

        // Create a card DOM element
        function createCardElement(card) {
            const cardElement = document.createElement('div');
            cardElement.className = 'kanban-card';
            cardElement.draggable = true;
            cardElement.dataset.cardId = card.id;
            cardElement.dataset.columnId = card.columnId;
            
            // Card content
            const contentDiv = document.createElement('div');
            contentDiv.className = 'kanban-card-content';
            contentDiv.textContent = card.content;
            
            // Card actions
            const actionsDiv = document.createElement('div');
            actionsDiv.className = 'kanban-card-actions';
            
            const editBtn = document.createElement('button');
            editBtn.className = 'kanban-edit-btn';
            editBtn.innerHTML = '<svg xmlns="http://www.w3.org/2000/svg" class="h-4 w-4" fill="none" viewBox="0 0 24 24" stroke="currentColor"><path stroke-linecap="round" stroke-linejoin="round" stroke-width="2" d="M11 5H6a2 2 0 00-2 2v11a2 2 0 002 2h11a2 2 0 002-2v-5m-1.414-9.414a2 2 0 112.828 2.828L11.828 15H9v-2.828l8.586-8.586z" /></svg>';
            editBtn.title = 'Edit card';
            editBtn.addEventListener('click', () => startEditingCard(card));
            
            const deleteBtn = document.createElement('button');
            deleteBtn.className = 'kanban-delete-btn';
            deleteBtn.innerHTML = '<svg xmlns="http://www.w3.org/2000/svg" class="h-4 w-4" fill="none" viewBox="0 0 24 24" stroke="currentColor"><path stroke-linecap="round" stroke-linejoin="round" stroke-width="2" d="M19 7l-.867 12.142A2 2 0 0116.138 21H7.862a2 2 0 01-1.995-2.158L5 7m5 4v6m4-6v6m1-10V4a1 1 0 00-1-1h-4a1 1 0 00-1 1v3M4 7h16" /></svg>';
            deleteBtn.title = 'Delete card';
            deleteBtn.addEventListener('click', () => confirmCardDeletion(card));
            
            actionsDiv.appendChild(editBtn);
            actionsDiv.appendChild(deleteBtn);
            
            cardElement.appendChild(contentDiv);
            cardElement.appendChild(actionsDiv);
            
            // Make the card draggable
            cardElement.addEventListener('dragstart', handleDragStart);
            cardElement.addEventListener('dragend', handleDragEnd);
            
            return cardElement;
        }

        // Create "Add card" button for a column
        function createAddCardButton(columnId) {
            const buttonDiv = document.createElement('div');
            buttonDiv.className = 'mt-4';
            
            const button = document.createElement('button');
            button.className = 'kanban-add-btn w-full';
            button.textContent = '+ Add Card';
            button.addEventListener('click', () => addNewCard(columnId));
            
            buttonDiv.appendChild(button);
            return buttonDiv;
        }

        // Update card counts for each column
        function updateCardCounts(cards) {
            COLUMNS.forEach(column => {
                const countElement = document.getElementById(`count-${column.id}`);
                const count = cards.filter(card => card.columnId === column.id).length;
                countElement.textContent = count;
            });
        }

        // Drag and drop handlers
        let draggedCard = null;

        function handleDragStart(e) {
            draggedCard = this;
            e.dataTransfer.effectAllowed = 'move';
            // Add a visual cue for dragging
            this.classList.add('opacity-70');
        }

        function handleDragEnd() {
            this.classList.remove('opacity-70');
        }

        // Handle drag over and drop events
        document.addEventListener('dragover', function(e) {
            e.preventDefault();
            const column = this.closest('.kanban-column');
            if (column) {
                // Visual feedback for drop target
                column.classList.add('bg-indigo-100', 'dark:bg-indigo-900');
            }
        });

        document.addEventListener('dragleave', function() {
            const columns = document.querySelectorAll('.kanban-column');
            columns.forEach(col => col.classList.remove('bg-indigo-100', 'dark:bg-indigo-900'));
        });

        document.addEventListener('drop', function(e) {
            e.preventDefault();
            const columns = document.querySelectorAll('.kanban-column');
            columns.forEach(col => col.classList.remove('bg-indigo-100', 'dark:bg-indigo-900'));
            
            const targetColumn = this.closest('.kanban-column');
            if (!targetColumn || !draggedCard) return;
            
            // Get column ID from the target element
            const newColumnId = targetColumn.id.replace('column-', '');
            
            // Get card data
            const cardId = draggedCard.dataset.cardId;
            const oldColumnId = draggedCard.dataset.columnId;
            
            // If moving to the same column, no change needed
            if (newColumnId === oldColumnId) return;
            
            // Update card data in localStorage
            const cards = JSON.parse(localStorage.getItem(KANBAN_NAMESPACE) || '[]');
            const cardIndex = cards.findIndex(card => card.id === cardId);
            
            if (cardIndex !== -1) {
                // Update column ID
                cards[cardIndex].columnId = newColumnId;
                
                // Save updated data
                saveToLocalStorage(cards);
                
                // Re-render the board to reflect changes
                renderBoard(cards);
                
                // Trigger confetti if moved to "Done" column
                if (newColumnId === 'done') {
                    triggerConfetti();
                }
            }
            
            draggedCard = null;
        });

        // Start editing a card in place
        function startEditingCard(card) {
            // Cancel any ongoing edit
            if (currentEditingCard) {
                cancelEditing();
            }
            
            // Find the card DOM element
            const cardElement = document.querySelector(`[data-card-id="${card.id}"]`);
            if (!cardElement) return;
            
            // Get the content div
            const contentDiv = cardElement.querySelector('.kanban-card-content');
            
            // Create input field
            const input = document.createElement('textarea');
            input.className = 'w-full p-2 border rounded bg-white dark:bg-gray-700 text-gray-800 dark:text-white';
            input.value = card.content;
            input.rows = 3;
            
            // Replace content with input
            contentDiv.innerHTML = '';
            contentDiv.appendChild(input);
            
            // Focus and select text
            input.focus();
            input.select();
            
            // Save reference to editing state
            currentEditingCard = card;
            currentEditingInput = input;
            
            // Add event listeners for saving/canceling
            const saveButton = document.createElement('button');
            saveButton.className = 'mt-2 px-3 py-1 bg-green-500 text-white rounded text-sm';
            saveButton.textContent = 'Save';
            
            const cancelButton = document.createElement('button');
            cancelButton.className = 'mt-2 ml-2 px-3 py-1 bg-gray-500 text-white rounded text-sm';
            cancelButton.textContent = 'Cancel';
            
            const actionsDiv = cardElement.querySelector('.kanban-card-actions');
            actionsDiv.innerHTML = '';
            actionsDiv.appendChild(saveButton);
            actionsDiv.appendChild(cancelButton);
            
            // Add event listeners for save/cancel
            saveButton.addEventListener('click', () => saveCardEdit(card, input.value));
            cancelButton.addEventListener('click', cancelEditing);
            
            // Handle Enter key (Save) and Escape key (Cancel)
            input.addEventListener('keydown', function(e) {
                if (e.key === 'Enter' && e.ctrlKey) {
                    saveCardEdit(card, input.value);
                } else if (e.key === 'Escape') {
                    cancelEditing();
                }
            });
        }

        // Save the edited card content
        function saveCardEdit(card, newContent) {
            if (!newContent.trim()) {
                alert('Card content cannot be empty!');
                return;
            }
            
            // Update in localStorage
            const cards = JSON.parse(localStorage.getItem(KANBAN_NAMESPACE) || '[]');
            const cardIndex = cards.findIndex(c => c.id === card.id);
            
            if (cardIndex !== -1) {
                cards[cardIndex].content = newContent;
                saveToLocalStorage(cards);
                
                // Re-render to reflect changes
                renderBoard(cards);
            }
            
            cancelEditing();
        }

        // Cancel ongoing card edit
        function cancelEditing() {
            if (currentEditingCard) {
                // Re-render the board to restore original state
                const cards = JSON.parse(localStorage.getItem(KANBAN_NAMESPACE) || '[]');
                renderBoard(cards);
                
                currentEditingCard = null;
                currentEditingInput = null;
            }
        }

        // Confirm card deletion
        function confirmCardDeletion(card) {
            // Create confirmation modal in the DOM (no window.alert)
            const modal = document.createElement('div');
            modal.className = 'fixed inset-0 bg-black bg-opacity-50 flex items-center justify-center z-50';
            modal.innerHTML = `
                <div class="bg-white dark:bg-gray-800 p-6 rounded-lg shadow-xl w-96">
                    <h3 class="text-lg font-semibold text-gray-800 dark:text-white mb-2">Delete Card</h3>
                    <p class="text-gray-600 dark:text-gray-300 mb-4">Are you sure you want to delete this card?</p>
                    <div class="flex justify-end space-x-3">
                        <button id="cancel-delete" class="px-4 py-2 bg-gray-200 dark:bg-gray-700 text-gray-800 dark:text-white rounded-md hover:bg-gray-300 dark:hover:bg-gray-600 transition-colors">
                            Cancel
                        </button>
                        <button id="confirm-delete" class="px-4 py-2 bg-red-500 text-white rounded-md hover:bg-red-600 transition-colors">
                            Delete
                        </button>
                    </div>
                </div>
            `;
            
            // Add modal to DOM
            document.body.appendChild(modal);
            
            // Handle actions
            const confirmBtn = modal.querySelector('#confirm-delete');
            const cancelBtn = modal.querySelector('#cancel-delete');
            
            confirmBtn.addEventListener('click', () => {
                deleteCard(card);
                document.body.removeChild(modal);
            });
            
            cancelBtn.addEventListener('click', () => {
                document.body.removeChild(modal);
            });
        }

        // Delete a card from localStorage and re-render
        function deleteCard(card) {
            const cards = JSON.parse(localStorage.getItem(KANBAN_NAMESPACE) || '[]');
            const filteredCards = cards.filter(c => c.id !== card.id);
            
            saveToLocalStorage(filteredCards);
            renderBoard(filteredCards);
        }

        // Add a new card to a column
        function addNewCard(columnId) {
            const content = prompt('Enter card content:');
            if (!content || !content.trim()) return;
            
            // Create new card object
            const newCard = {
                id: Date.now().toString(), // Simple unique ID generator
                content: content.trim(),
                columnId: columnId
            };
            
            // Load current cards and add new one
            const cards = JSON.parse(localStorage.getItem(KANBAN_NAMESPACE) || '[]');
            cards.push(newCard);
            
            saveToLocalStorage(cards);
            renderBoard(cards);
        }

        // Trigger confetti animation for "Done" column
        function triggerConfetti() {
            // Simple confetti effect using emojis (can be replaced with a proper library)
            const container = document.querySelector('#kanban-board');
            
            // Create confetti particles
            for (let i = 0; i < 50; i++) {
                const confetti = document.createElement('div');
                confetti.className = 'absolute text-xl animate-bounce';
                confetti.textContent = ['🎉', '🎊', '✨', '⭐'][Math.floor(Math.random() * 4)];
                confetti.style.left = `${Math.random() * 100}%`;
                confetti.style.top = '-20px';
                confetti.style.animationDuration = `${Math.random() * 2 + 1}s`;
                
                container.appendChild(confetti);
                
                // Remove element after animation
                setTimeout(() => {
                    confetti.remove();
                }, 3000);
            }
        }

        // Initialize the board when DOM is ready
        document.addEventListener('DOMContentLoaded', initKanbanBoard);
    </script>
</body>
