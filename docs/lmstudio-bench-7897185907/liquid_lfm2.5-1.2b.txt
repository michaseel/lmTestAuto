<script>
  // Configuration
  const COLUMNS = ['Backlog', 'In Progress', 'Review', 'Done'];
  const COLUMNS_NAMES = ['Backlog', 'In Progress', 'Review', 'Done'];
  const NUM_CARDS = 10; // Number of cards
  let cards = [];
  const STORAGE_KEY = 'kanban_cards';
  const COLUMN_NAMES = COLUMNS;
  const COLUMN_IDX = { 'Backlog': 0, 'In Progress': 1, 'Review': 2, 'Done': 3 };

  // Load cards from localStorage or create if empty
  function loadCards() {
    const data = JSON.parse(localStorage.getItem(STORAGE_KEY)) || [];
    data.forEach(id => cards.push({ id, name: `Card ${id}` }));
    render();
  }

  // Save cards to localStorage
  function saveCards() {
    localStorage.setItem(STORAGE_KEY, JSON.stringify(cards));
  }

  // Render cards on screen
  function render() {
    const container = document.querySelector('.kanban-container');
    container.innerHTML = '';

    cards.forEach((card, index) => {
      const col = COLUMNS_NAMES[index];
      const cardElement = document.createElement('div');
      cardElement.className = `card ${card.done ? 'done' : ''}`;
      cardElement.draggable = true;

      // Card title
      const title = document.createElement('div');
      title.className = 'card-title';
      title.textContent = card.name;
      cardElement.appendChild(title);

      // Card content
      const content = document.createElement('div');
      content.className = 'card-content';
      cardElement.appendChild(content);

      // Edit button
      const editBtn = document.createElement('button');
      editBtn.className = 'card-actions';
      editBtn.textContent = 'Edit';
      editBtn.onclick = () => editCard(card, content);
      cardElement.appendChild(editBtn);

      // Add card
      if (index === NUM_CARDS - 1) {
        cardElement.textContent = '✅ Done';
      }

      // Add card action
      const addBtn = document.createElement('button');
      addBtn.className = 'card-actions';
      addBtn.textContent = 'Add';
      cardElement.appendChild(addBtn);

      // Delete button
      const deleteBtn = document.createElement('button');
      deleteBtn.className = 'card-actions';
      deleteBtn.textContent = 'Delete';
      deleteBtn.onclick = () => {
        if (confirm('Are you sure you want to delete this card?')) {
          cards.splice(index, 1);
          saveCards();
          render();
        }
      };
      cardElement.appendChild(deleteBtn);

      container.appendChild(cardElement);
    });
  }

  // Load cards on page load
  loadCards();

  // Add a card to any column
  function addCard(event) {
    const col = event.target.dataset.column;
    const name = `Card ${cards.length + 1}`;
    const card = { id: Date.now(), name };
    cards.push(card);
    saveCards();
    render();
    event.target.classList.add('dragging');
  }

  // Drag and drop logic
  document.addEventListener('dragstart', (e) => {
    e.target.classList.add('dragging');
  });

  document.addEventListener('dragend', () => {
    e.target.classList.remove('dragging');
  });

  document.addEventListener('dragover', (e) => {
    e.preventDefault();
  });

  document.addEventListener('drop', (e) => {
    e.preventDefault();
    const dragId = e.dataTransfer.filename;
    const source = e.target.closest('.column');
    const targetCol = COLUMNS_NAMES[e.target.dataset.column];
    const card = cards.find(c => c.id === parseInt(source.dataset.id));

    if (source && targetCol) {
      const col = COLUMNS_NAMES[targetCol];
      if (cards.some(c => c.id === card.id)) {
        const index = col.indexOf(targetCol);
        if (index !== -1) {
          cards.splice(index, 1); // Remove from old column
          cards.unshift(card); // Insert at front
          saveCards();
          render();
        }
      }
    }
  });

  // Edit card
  function editCard(card, content) {
    const col = COLUMNS_NAMES[card.dataset.column];
    card.id = card.name;
    content.textContent = card.name;
  }

  // Delete card
  function deleteCard(button) {
    const col = button.dataset.column;
    cards.splice(card.id, 1);
    saveCards();
    render();
  }

  // Confirm delete
  document.querySelectorAll('.card-actions').forEach(btn => {
    btn.addEventListener('click', () => deleteCard(btn));
  });

  // Initialize UI
  document.addEventListener('DOMContentLoaded', render);
</script>

</body>
</html>
