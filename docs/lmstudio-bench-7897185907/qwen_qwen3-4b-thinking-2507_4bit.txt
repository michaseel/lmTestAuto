function init() {
  // Load state

  } else {

  }

function createDummyCards() {
  const dummyCards = [
    { id: 'dummy1', text: 'Dummy task 1', column: 'backlog' },
    { id: 'dummy2', text: 'Dummy task 2', column: 'backlog' },
    { id: 'dummy3', text: 'Dummy task 3', column: 'in-progress' }

  localStorage.setItem(CANBAN_STATE_KEY, JSON.stringify(dummyCards));
  renderCards(dummyCards);
}

function renderCards(state) {
  // Clear existing cards
  const columns = document.querySelectorAll('.column');
  columns.forEach(column => {
    const columnId = column.dataset.column;
    const cardsContainer = document.getElementById(`${columnId}-cards`);
    const cards = state.filter(card => card.column === columnId);
    cardsContainer.innerHTML = '';
    cards.forEach(card => {
      const cardElement = document.createElement('div');
      cardElement.className = 'card bg-white rounded-lg shadow-md p-4 transition-all duration-300 hover:shadow-lg';
      cardElement.innerHTML = `
        <div class="text-lg font-semibold">${card.text}</div>
        <div class="flex space-x-2 mt-2">
          <button class="delete-btn btn" data-card-id="${card.id}">Delete</button>
          <button class="edit-btn btn" data-card-id="${card.id}">Edit</button>
        </div>
      `;
      cardsContainer.appendChild(cardElement);
    });
  });
}

function toggleEdit(cardElement) {
  const textElement = cardElement.querySelector('.card-text');
  const input = document.createElement('input');
  input.className = 'edit-input';
  input.type = 'text';
  input.value = textElement.textContent;
  input.style.display = 'block';
  input.style.opacity = '0';
  input.style.transition = 'opacity 0.3s';
  input.style.position = 'relative';
  input.style.zIndex = '100';

  // Add event listener for input and blur
  input.addEventListener('input', (e) => {
    textElement.textContent = e.target.value;
  });
  input.addEventListener('blur', () => {
    // Save the text
    const newText = input.value;
    // Update the card's text
    textElement.textContent = newText;
    // Hide the input
    input.style.opacity = '0';
    input.style.display = 'none';
  });

  // Show the input
  textElement.parentNode.insertBefore(input, textElement);
  input.style.opacity = '1';
  input.style.display = 'block';
}
