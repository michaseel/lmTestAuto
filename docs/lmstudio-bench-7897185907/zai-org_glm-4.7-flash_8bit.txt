    <script>
        // --- State Management ---
        const STORAGE_KEY = 'vibrant_kanban_data_v1';
        let tasks = [];
        let currentModalMode = 'add'; // 'add' or 'edit'
        let currentTargetColumn = '';
        let currentEditId = null;
        let deleteTargetId = null;

        // --- Initialization ---
        document.addEventListener('DOMContentLoaded', () => {
            loadTasks();
            renderBoard();
            setupDragAndDrop();
        });

        function loadTasks() {
            const stored = localStorage.getItem(STORAGE_KEY);
            if (stored) {
                tasks = JSON.parse(stored);
            } else {
                // Initialize with dummy data if empty
                tasks = [
                    { id: 't1', col: 'backlog', content: 'Research new design trends' },
                    { id: 't2', col: 'inprogress', content: 'Setup project repository' },
                    { id: 't3', col: 'review', content: 'Review pull request #42' },
                    { id: 't4', col: 'done', content: 'Initial project setup' },
                    { id: 't5', col: 'backlog', content: 'Write documentation' }
                ];
                saveTasks();
            }
        }

        function saveTasks() {
            localStorage.setItem(STORAGE_KEY, JSON.stringify(tasks));
            updateCounts();
        }

        // --- Rendering ---
        function renderBoard() {
            // Clear all columns
            ['backlog', 'inprogress', 'review', 'done'].forEach(colId => {
                const colEl = document.getElementById(`col-${colId}`);
                colEl.innerHTML = '';
            });

            // Populate columns
            tasks.forEach(task => {
                const card = createCardElement(task);
                document.getElementById(`col-${task.col}`).appendChild(card);
            });

            updateCounts();
        }

        function createCardElement(task) {
            const div = document.createElement('div');
            div.className = 'kanban-card bg-white p-4 rounded-lg border border-gray-200 select-none group';
            div.draggable = true;
            div.dataset.id = task.id;

            // Inner HTML
            div.innerHTML = `
                <p class="text-slate-700 text-sm leading-relaxed break-words">${escapeHtml(task.content)}</p>
                <div class="mt-3 flex justify-between items-center opacity-0 group-hover:opacity-100 transition-opacity">
                    <button onclick="openEditModal('${task.id}')" class="text-xs text-indigo-500 hover:text-indigo-700 font-medium">Edit</button>
                    <button onclick="openDeleteModal('${task.id}')" class="text-xs text-red-400 hover:text-red-600 font-medium">Delete</button>
                </div>
            `;

            // Double-click to edit
            div.addEventListener('dblclick', () => openEditModal(task.id));

            return div;
        }

        function updateCounts() {
            const counts = {
                backlog: tasks.filter(t => t.col === 'backlog').length,
                inprogress: tasks.filter(t => t.col === 'inprogress').length,
                review: tasks.filter(t => t.col === 'review').length,
                done: tasks.filter(t => t.col === 'done').length
            };

            for (const [key, value] of Object.entries(counts)) {
                document.getElementById(`count-${key}`).textContent = value;
            }
        }

        // --- Drag and Drop Logic ---
        let draggedItem = null;

        function setupDragAndDrop() {
            const columns = document.querySelectorAll('.kanban-column');
            
            columns.forEach(col => {
                // Add drop event to column
                col.addEventListener('dragover', (e) => {
                    e.preventDefault(); // Allow dropping
                    col.classList.add('bg-indigo-50/50');
                });

                col.addEventListener('dragleave', () => {
                    col.classList.remove('bg-indigo-50/50');
                });

                col.addEventListener('drop', (e) => {
                    e.preventDefault();
                    col.classList.remove('bg-indigo-50/50');
                    
                    if (draggedItem) {
                        const newColId = col.dataset.col;
                        const taskId = draggedItem.dataset.id;
                        
                        // Update State
                        const taskIndex = tasks.findIndex(t => t.id === taskId);
                        if (taskIndex > -1) {
                            const oldCol = tasks[taskIndex].col;
                            tasks[taskIndex].col = newColId;
                            
                            // If moved to Done, trigger confetti
                            if (oldCol !== 'done' && newColId === 'done') {
                                fireConfetti();
                            }

                            saveTasks();
                            renderBoard();
                        }
                        draggedItem = null;
                    }
                });
            });

            // Add drag events to cards

        // --- Modal & Form Logic ---
        const modalBackdrop = document.getElementById('modal-backdrop');
        const formModal = document.getElementById('form-modal');
        const deleteModal = document.getElementById('delete-modal');
        const taskInput = document.getElementById('task-input');
        const errorMsg = document.getElementById('error-msg');

        function openAddCardModal(colId) {
            currentModalMode = 'add';
            currentTargetColumn = colId;
            currentEditId = null;
            
            document.getElementById('modal-title').textContent = 'New Task';
            document.getElementById('submit-btn-text').textContent = 'Add Task';
            taskInput.value = '';
            errorMsg.classList.add('hidden');
            
            showModals();
        }

        function openEditModal(taskId) {
            currentModalMode = 'edit';
            currentEditId = taskId;
            const task = tasks.find(t => t.id === taskId);
            
            if (!task) return;

            document.getElementById('modal-title').textContent = 'Edit Task';
            document.getElementById('submit-btn-text').textContent = 'Save Changes';
            taskInput.value = task.content;
            errorMsg.classList.add('hidden');
            
            showModals();
        }

        function openDeleteModal(taskId) {
            deleteTargetId = taskId;
            hideModals(); // Hide form modal first
            deleteModal.classList.remove('hidden');
            modalBackdrop.classList.remove('hidden');
        }

        function showModals() {
            hideModals();
            formModal.classList.remove('hidden');
            modalBackdrop.classList.remove('hidden');
            setTimeout(() => taskInput.focus(), 100);
        }

        function hideModals() {
            modalBackdrop.classList.add('hidden');
            formModal.classList.add('hidden');
            deleteModal.classList.add('hidden');
        }

        function closeModal() {
            hideModals();
            deleteTargetId = null;
            currentEditId = null;
        }

        function handleFormSubmit() {
            const content = taskInput.value.trim();
            if (!content) {
                errorMsg.classList.remove('hidden');
                return;
            }

            if (currentModalMode === 'add') {
                const newTask = {
                    id: 't' + Date.now(),
                    col: currentTargetColumn,
                    content: content
                };
                tasks.push(newTask);
            } else if (currentModalMode === 'edit') {
                const taskIndex = tasks.findIndex(t => t.id === currentEditId);
                if (taskIndex > -1) {
                    tasks[taskIndex].content = content;
                }
            }

            saveTasks();
            renderBoard();
            closeModal();
        }

        function confirmDelete() {
            if (deleteTargetId) {
                tasks = tasks.filter(t => t.id !== deleteTargetId);
                saveTasks();
                renderBoard();
                closeModal();
            }
        }

        // --- Utility ---
        function escapeHtml(text) {
            const map = {
                '&': '&amp;',
                '<': '&lt;',
                '>': '&gt;',
                '"': '&quot;',
                "'": '&#039;'
            };
            return text.replace(/[&<>"']/g, function(m) { return map[m]; });
        }

        // --- Confetti Effect (Simple Canvas Implementation) ---
        const canvas = document.getElementById('confetti-canvas');
        const ctx = canvas.getContext('2d');
        let particles = [];

        function resizeCanvas() {
            canvas.width = window.innerWidth;
            canvas.height = window.innerHeight;
        }
        window.addEventListener('resize', resizeCanvas);
        resizeCanvas();

        function fireConfetti() {
            const colors = ['#ef4444', '#3b82f6', '#10b981', '#f59e0b', '#8b5cf6'];
            for (let i = 0; i < 150; i++) {
                particles.push({
                    x: window.innerWidth / 2,
                    y: window.innerHeight / 2,
                    vx: (Math.random() - 0.5) * 20,
                    vy: (Math.random() - 1) * 20 - 5,
                    size: Math.random() * 8 + 4,
                    color: colors[Math.floor(Math.random() * colors.length)],
                    life: 100
                });
            }
            if (particles.length <= 150) {
                requestAnimationFrame(updateConfetti);
            }
        }

        function updateConfetti() {
            ctx.clearRect(0, 0, canvas.width, canvas.height);
            
            for (let i = 0; i < particles.length; i++) {
                const p = particles[i];
                p.x += p.vx;
                p.y += p.vy;
                p.vy += 0.5; // gravity
                p.life--;
                
                ctx.fillStyle = p.color;
                ctx.fillRect(p.x, p.y, p.size, p.size);
            }
            
            particles = particles.filter(p => p.life > 0);
            
            if (particles.length > 0) {
                requestAnimationFrame(updateConfetti);
            }
        }
